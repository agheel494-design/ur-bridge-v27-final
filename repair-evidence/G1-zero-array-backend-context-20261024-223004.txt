/** Emits a forward declaration (prototype) for a function, used so call order in C never matters (matches UR's own forward-declaration semantics). */
function emitPrototype(fn: IRFunction): string {
  const params = fn.params.flatMap(p => p.structLayout
    ? [`struct ${cStructTypeName(p.structLayout.structName)}`]
    : p.arrayLength !== undefined
    ? [`${cType(p.type)}[]`, `int64_t`]
    : [cType(p.type)]
  ).join(", ") || "void";
  const returnTypeStr = fn.returnStructLayout ? `struct ${cStructTypeName(fn.returnStructLayout.structName)}` : cType(fn.returnType);
  return `${returnTypeStr} ${fn.qualifiedName}(${params});`;
}

/**

    new Uint8Array([OPCODE_I32_ADD]),

}

/** Encodes the module's `global` section: exactly one mutable `i32` global (the bump pointer), initialized to `BUMP_INITIAL_ADDRESS`. */
export function encodeGlobalSection(): Uint8Array {
  const SECTION_GLOBAL = 6;
  const VALTYPE_I32 = 0x7f;
  const MUTABLE = 0x01;
  const globalEntry = concatBytes(
    new Uint8Array([VALTYPE_I32, MUTABLE, OPCODE_I32_CONST]), encodeSLEB128(BigInt(BUMP_INITIAL_ADDRESS)), new Uint8Array([OPCODE_END]),
  );
  const payload = concatBytes(encodeULEB128(1), globalEntry);
  return concatBytes(new Uint8Array([SECTION_GLOBAL]), encodeULEB128(payload.length), payload);
}
