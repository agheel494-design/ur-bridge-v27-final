describe("WASM Backend - ARRAY_LAYOUT_CONTRACT (permanent regression guard, per WASM_MEMORY_MODEL_SPEC.md section 9 - directly inspects real memory contents, not merely the final computed result)", () => {
  async function getMemoryAfterExecution(src: string): Promise<DataView> {
    const module = compileToModule(src);
    const bytes = emitWasmModule(module);
    const { instance } = await WebAssembly.instantiate(bytes);
    (instance.exports.main as () => bigint)();
    return new DataView((instance.exports.memory as WebAssembly.Memory).buffer);
  }

  it("a single 3-element array lands at EXACTLY the offsets WASM_MEMORY_MODEL_SPEC.md section 2 specifies (length at +0, elements at +8/+16/+24 relative to base)", async () => {
    const view = await getMemoryAfterExecution("function main(): int\n  let arr = [10, 20, 30]\n  return arr[1]\nend\n");
    // Base address is BUMP_INITIAL_ADDRESS (8), per the Memory Model.
    expect(view.getBigInt64(8, true)).toBe(3n);   // length
    expect(view.getBigInt64(16, true)).toBe(10n); // arr[0]
    expect(view.getBigInt64(24, true)).toBe(20n); // arr[1]
    expect(view.getBigInt64(32, true)).toBe(30n); // arr[2]
  });

  it("the exact two-array layout from MEMORY_LAYOUT_EXAMPLES.md Example 2 (adjusted for this Backend's actual BUMP_INITIAL_ADDRESS base of 8, not the example's illustrative 1024)", async () => {
    const view = await getMemoryAfterExecution("function main(): int\n  let a = [7, 8]\n  let b = [1, 2, 3, 4]\n  return b[2]\nend\n");
    expect(view.getBigInt64(8, true)).toBe(2n);   // a.length
    expect(view.getBigInt64(16, true)).toBe(7n);  // a[0]
    expect(view.getBigInt64(24, true)).toBe(8n);  // a[1]
    expect(view.getBigInt64(32, true)).toBe(4n);  // b.length (a's base 8 + a's total size 24 = 32)
    expect(view.getBigInt64(40, true)).toBe(1n);  // b[0]
    expect(view.getBigInt64(48, true)).toBe(2n);  // b[1]
    expect(view.getBigInt64(56, true)).toBe(3n);  // b[2]
    expect(view.getBigInt64(64, true)).toBe(4n);  // b[3]
  });

  it("address 0-7 (the reserved zero region) is never written by any array allocation", async () => {
    const view = await getMemoryAfterExecution("function main(): int\n  let arr = [10, 20, 30]\n  return arr[1]\nend\n");
    expect(view.getBigInt64(0, true)).toBe(0n);
  });
});
